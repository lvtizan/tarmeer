# Arabic i18n + Auto-Translation Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Add Arabic language switching to Tarmeer with auto-translation of all dynamic content via Azure Translator Free Tier.

**Architecture:** Language preference stored in localStorage + React Context, setting `dir="rtl"` on `<html>`. Dynamic content (company descriptions, project descriptions) pre-translated at save time and stored in `_ar` DB columns. Static UI strings translated via a global `t()` helper. Batch script translates all existing data once.

**Tech Stack:** Azure Translator REST API (free, 2M chars/month), MySQL column additions via existing autoMigrate pattern, React Context, Tailwind `rtl:` variants, Cairo font (Google Fonts).

## Task 1: Backend Translate Utility

**Files:**
- Create: `server/src/lib/translate.ts`

**Step 1: Create the utility**

```typescript
// server/src/lib/translate.ts

const AZURE_KEY = process.env.AZURE_TRANSLATOR_KEY;
const AZURE_REGION = process.env.AZURE_TRANSLATOR_REGION || 'global';
const ENDPOINT = 'https://api.cognitive.microsofttranslator.com/translate?api-version=3.0&to=ar';

/**
 * Translate multiple text fields to Arabic in a single API call.
 * Returns null for each field if translation fails or key not configured.
 * Never throws — translation failure must not block the main flow.
 *
 * Usage:
 *   const ar = await translateFields({ title, description });
 *   // ar.title, ar.description
 */
export async function translateFields(
  fields: Record<string, string | null | undefined>
): Promise<Record<string, string | null>> {
  const nullResult = () =>
    Object.fromEntries(Object.keys(fields).map((k) => [k, null]));

  if (!AZURE_KEY) return nullResult();

  const entries = Object.entries(fields).filter(([, v]) => v?.trim());
  if (!entries.length) return nullResult();

  try {
    const res = await fetch(ENDPOINT, {
      method: 'POST',
      headers: {
        'Ocp-Apim-Subscription-Key': AZURE_KEY,
        'Ocp-Apim-Subscription-Region': AZURE_REGION,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify(entries.map(([, v]) => ({ text: v!.trim() }))),
    });

    if (!res.ok) {
      console.warn('[translate] Azure API error:', res.status);
      return nullResult();
    }

    const data = await res.json();
    const result: Record<string, string | null> = {};
    entries.forEach(([key], i) => {
      result[key] = data[i]?.translations?.[0]?.text ?? null;
    });
    return result;
  } catch (err) {
    console.warn('[translate] Failed:', err);
    return nullResult();
  }
}
```

## Task 5: Batch Translate Existing Data

**Files:**
- Create: `scripts/batch-translate.mjs`

This is a one-time script. Run it once, then archive it.

```javascript
// scripts/batch-translate.mjs
// Usage: AZURE_TRANSLATOR_KEY=xxx AZURE_TRANSLATOR_REGION=uaenorth node scripts/batch-translate.mjs

import mysql from 'mysql2/promise';
import { readFileSync } from 'fs';
import { fileURLToPath } from 'url';
import path from 'path';

// Load .env from server/
const envPath = path.join(path.dirname(fileURLToPath(import.meta.url)), '../server/.env');
const env = Object.fromEntries(
  readFileSync(envPath, 'utf8').split('\n')
    .filter(l => l.includes('=') && !l.startsWith('#'))
    .map(l => l.split('=').map(s => s.trim()))
);

const AZURE_KEY = process.env.AZURE_TRANSLATOR_KEY || env.AZURE_TRANSLATOR_KEY;
const AZURE_REGION = process.env.AZURE_TRANSLATOR_REGION || env.AZURE_TRANSLATOR_REGION || 'global';
const ENDPOINT = 'https://api.cognitive.microsofttranslator.com/translate?api-version=3.0&to=ar';

const pool = await mysql.createPool({
  host: env.DB_HOST, user: env.DB_USER, password: env.DB_PASSWORD,
  database: env.DB_NAME, port: parseInt(env.DB_PORT || '3306'),
});

async function translateBatch(texts) {
  if (!texts.length) return [];
  const res = await fetch(ENDPOINT, {
    method: 'POST',
    headers: {
      'Ocp-Apim-Subscription-Key': AZURE_KEY,
      'Ocp-Apim-Subscription-Region': AZURE_REGION,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify(texts.map(t => ({ text: t || '' }))),
  });
  if (!res.ok) throw new Error(`Azure error: ${res.status}`);
  const data = await res.json();
  return data.map(d => d.translations[0].text);
}

async function batchTable({ table, idCol, fields, nullCheck }) {
  console.log(`\n=== ${table} ===`);
  const [rows] = await pool.query(
    `SELECT ${idCol}, ${fields.join(', ')} FROM ${table} WHERE ${nullCheck} IS NULL`
  );
  console.log(`Found ${rows.length} rows to translate`);

  for (const row of rows) {
    try {
      const texts = fields.map(f => row[f] || '');
      const translated = await translateBatch(texts);
      const setClause = fields.map((f, i) => `${f}_ar = ?`).join(', ');
      await pool.query(
        `UPDATE ${table} SET ${setClause} WHERE ${idCol} = ?`,
        [...translated, row[idCol]]
      );
      process.stdout.write('.');
    } catch (err) {
      console.error(`\nFailed row ${row[idCol]}:`, err.message);
    }
    // Respect rate limits
    await new Promise(r => setTimeout(r, 100));
  }
  console.log('\nDone');
}

await batchTable({
  table: 'company_profiles',
  idCol: 'id',
  fields: ['company_name', 'tagline', 'description'],
  nullCheck: 'description_ar',
});

await batchTable({
  table: 'projects',
  idCol: 'id',
  fields: ['title', 'description'],
  nullCheck: 'description_ar',
});

await batchTable({
  table: 'uae_companies',
  idCol: 'id',
  fields: ['name', 'description'],
  nullCheck: 'description_ar',
});

await pool.end();
console.log('\nBatch translation complete.');
```

## Task 6: Frontend LanguageContext

**Files:**
- Create: `src/contexts/LanguageContext.tsx`
- Modify: `src/main.tsx` — wrap app in LanguageProvider

**Step 1: Create context**

```tsx
// src/contexts/LanguageContext.tsx
import { createContext, useContext, useState, useEffect, type ReactNode } from 'react';

export type Lang = 'en' | 'ar';

interface LanguageContextValue {
  lang: Lang;
  setLang: (lang: Lang) => void;
  isAr: boolean;
}

const LanguageContext = createContext<LanguageContextValue>({
  lang: 'en', setLang: () => {}, isAr: false,
});

export function LanguageProvider({ children }: { children: ReactNode }) {
  const [lang, setLangState] = useState<Lang>(
    () => (localStorage.getItem('tarmeer_lang') as Lang) || 'en'
  );

  const applyLang = (next: Lang) => {
    document.documentElement.dir = next === 'ar' ? 'rtl' : 'ltr';
    document.documentElement.lang = next;
  };

  const setLang = (next: Lang) => {
    setLangState(next);
    localStorage.setItem('tarmeer_lang', next);
    applyLang(next);
  };

  // Apply on first render
  useEffect(() => { applyLang(lang); }, []);

  return (
    <LanguageContext.Provider value={{ lang, setLang, isAr: lang === 'ar' }}>
      {children}
    </LanguageContext.Provider>
  );
}

export const useLang = () => useContext(LanguageContext);
```

/** Returns the Arabic value if available and lang=ar, otherwise falls back to English */
export function i18n(enValue: string | undefined, arValue: string | undefined, lang: Lang): string {
  if (lang === 'ar' && arValue?.trim()) return arValue;
  return enValue || '';
}
